二 建目录和工程文件
———————————————————————————————————————————————————————————————————————

建了include，src和config三个文件夹（本质上拆分各个功能）
工程

CMakeLists.txt
如下
{cmake_minimum_required(VERSION 3.25)
project(logkeeper VERSION 1.0.0 LANGUAGES CXX)#解释工程与版本

set(CMAKE_CXX_STANDARD 17)
add_compile_options(-finput-charset=UTF-8 -fexec-charset=GBK)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(logkeeper_lib src/logkeeper.cpp)
target_include_directories(logkeeper_lib PUBLIC include)#把 main.cpp 做成最终的程序.exe,链接一下其他的程序

add_executable(logkeeper src/main.cpp)
target_link_libraries(logkeeper PRIVATE logkeeper_lib)#链接}

CMakePresets.json（快捷指定编译器，版本和路径等）
如下
{
  "version": 3,
  "cmakeMinimumRequired": {
    "major": 3,
    "minor": 25
  },
  "configurePresets": [
    {
      "name": "gcc-x64-debug",
      "generator": "MinGW Makefiles",
      "binaryDir": "${sourceDir}/build/${presetName}",
      "cacheVariables": {
        "CMAKE_BUILD_TYPE": "Debug",
        "CMAKE_CXX_STANDARD": "17",
        "CMAKE_CXX_FLAGS": "-m64"
      }
    }
  ],
  "buildPresets": [
    {
      "name": "gcc-x64-debug",
      "configurePreset": "gcc-x64-debug"
    }
  ]
}
注意这里出了一个问题 "binaryDir": "${sourceDir}/build"  改成了  "binaryDir": "${sourceDir}/build/${presetName}"AI解释会导致缓存复用，改后缓存独立存储
